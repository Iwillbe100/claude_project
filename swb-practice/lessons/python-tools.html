dict와 set

count[word] = count.get(word, 0) + 1     # 없으면 0으로 시작
if word in count:                        # O(1)
    del count[word]
value = count.pop(word, -1)              # 있으면 꺼내고, 없으면 -1

from collections import defaultdict

heapq: 최소 힙

from heapq import heappush, heappop

heap = []
heappush(heap, (-score, user_id))        # 최대 힙은 부호를 뒤집는다
neg_score, uid = heappop(heap)           # 점수가 가장 큰 것, 같으면 ID가 작은 것
top = heap[0]                            # 꺼내지 않고 보기

튜플은 앞에서부터 비교하므로 (-점수, ID)로 넣으면 "점수 내림차순, 같으면 ID 오름차순"이 된다.

deque: 양쪽 끝이 O(1)인 큐

from collections import deque

q = deque([(0, 0)])
r, c = q.popleft()
q.append((r, c + 1))

bisect: 정렬된 리스트에서 위치 찾기

from bisect import bisect_left, bisect_right, insort

a = [1, 3, 3, 7]
bisect_left(a, 3)     # 1  (3 이상이 처음 나오는 위치)
bisect_right(a, 3)    # 3  (3 초과가 처음 나오는 위치)
insort(a, 5)          # 정렬을 유지하며 삽입 (삽입은 O(N)이지만 빠르다)

copy = [row[:] for row in mMap]      # 입력으로 받은 2차원 리스트는 이렇게 복사해 둔다

global: 함수 안에서 전역 변수에 대입할 때

count = 0
items = []

def init():
    global count, items      # 대입(=)하는 전역 변수는 선언해야 한다
    count = 0
    items = []

def add(x):
    global count
    count += 1               # += 도 대입이다
    items.append(x)          # 리스트 내용을 바꾸는 건 선언이 필요 없다
init에서 items = []만 쓰고 global을 빼먹으면 지역 변수만 새로 만들어지고 전역 리스트는 그대로 남는다. 가장 많이 틀리는 부분이다. 리스트는 items.clear()로 비우면 global 선언 없이도 된다.
